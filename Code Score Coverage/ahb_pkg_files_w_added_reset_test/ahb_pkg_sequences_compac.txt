class ahb_write_single_seq extends ahb_base_seq;
rand logic [31:0] addr;
rand logic [31:0] wdata;
rand bit          lock;
int unsigned      max_attempts = 1;
logic [1:0] resp;
constraint c_defaults {
soft lock      == 1'b0;
addr[1:0]      == 2'b00;
}
`uvm_object_utils(ahb_write_single_seq)
function new(string name = "ahb_write_single_seq");
super.new(name);
endfunction
task body();
ahb_seq_item it;
it = ahb_seq_item::type_id::create("w_it");
start_item(it);
it.dir          = AHB_WRITE;
it.addr         = addr;
it.wdata        = wdata;
it.trans        = T_NONSEQ;
it.burst        = B_SINGLE;
it.size         = SZ_WORD;
it.lock         = lock;
it.busreq       = 1'b1;
it.num_beats    = 1;
it.max_attempts = max_attempts;
it.exp_resp     = R_OKAY;
finish_item(it);
resp = it.resp;
endtask
endclass : ahb_write_single_seq

typedef enum {
PAT_DEAD,
PAT_INDEX,
PAT_RAND
} burst_pat_e;
class ahb_write_burst_seq extends ahb_base_seq;
rand logic [31:0]   base_addr;
rand logic [2:0]    burst_type;
burst_pat_e         pattern = PAT_DEAD;
logic [31:0]        user_data[$];
bit                 abort_aware  = 1'b0;
int unsigned        max_attempts = 1;
constraint c_align { base_addr[1:0] == 2'b00; }
constraint c_dflt  { soft burst_type == B_INCR4; }
`uvm_object_utils(ahb_write_burst_seq)
function new(string name = "ahb_write_burst_seq"); super.new(name); endfunction
function int unsigned beats_of(logic [2:0] b);
case (b)
B_INCR4, B_WRAP4  : return 4;
B_INCR8, B_WRAP8  : return 8;
B_INCR16,B_WRAP16 : return 16;
default           : return 1;
endcase
endfunction
task body();
ahb_seq_item it;
int unsigned N;
it = ahb_seq_item::type_id::create("wr_burst");
N  = beats_of(burst_type);
start_item(it);
it.dir       = AHB_WRITE;
it.addr      = base_addr;
it.trans     = T_NONSEQ;
it.burst     = burst_type;
it.size      = SZ_WORD;
it.lock      = 1'b0;
it.busreq    = 1'b1;
it.num_beats = N;
it.abort_aware  = abort_aware;
it.max_attempts = max_attempts;
it.beats_wr  = {};
if (user_data.size() == N) begin
it.beats_wr = user_data;
end else begin
for (int i = 0; i < N; i++) begin
case (pattern)
PAT_DEAD  : it.beats_wr.push_back(32'hDEAD_BE00 | i);
PAT_INDEX : it.beats_wr.push_back(32'hAAAA_0000 | i);
default   : it.beats_wr.push_back($urandom());
endcase
end
end
it.wdata    = it.beats_wr[0];
it.exp_resp = R_OKAY;
finish_item(it);
`uvm_info("SEQ_WR", $sformatf("burst done base=0x%08h burst=%0d N=%0d",
base_addr, burst_type, N), UVM_LOW)
endtask
endclass : ahb_write_burst_seq

class ahb_m3_smoke_vseq extends uvm_sequence;
`uvm_object_utils(ahb_m3_smoke_vseq)
`uvm_declare_p_sequencer(ahb_virtual_sequencer)
function new(string name = "ahb_m3_smoke_vseq");
super.new(name);
endfunction
task body();
ahb_write_single_seq w_seq;
ahb_read_single_seq  r_seq;
logic [31:0] initial_val;
`uvm_info("M3_SMOKE", "===== M3 Smoke Test Starting =====", UVM_LOW)
`uvm_info("M3_SMOKE", "Step 1: Read initial S1[1] (expect 0x14 = 20)", UVM_LOW)
r_seq = ahb_read_single_seq::type_id::create("r_seq_pre");
r_seq.addr = 32'h0000_0004;
r_seq.lock = 1'b0;
r_seq.start(p_sequencer.mst_seqr());
initial_val = r_seq.rdata;
`uvm_info("M3_SMOKE",
$sformatf("  → got 0x%08h, resp=%s",
initial_val, ahb_seq_item::resp_name(r_seq.resp)),
UVM_LOW)
if (initial_val !== 32'h0000_0014)
`uvm_warning("M3_SMOKE",
$sformatf("Initial S1[1] = 0x%08h, expected 0x14",
initial_val))
`uvm_info("M3_SMOKE", "Step 2: Write 0xDEADBEEF to S1[1]", UVM_LOW)
w_seq = ahb_write_single_seq::type_id::create("w_seq");
w_seq.addr  = 32'h0000_0004;
w_seq.wdata = 32'hDEAD_BEEF;
w_seq.lock  = 1'b0;
w_seq.start(p_sequencer.mst_seqr());
`uvm_info("M3_SMOKE", "Step 3: Read back S1[1] (expect 0xDEADBEEF)", UVM_LOW)
r_seq = ahb_read_single_seq::type_id::create("r_seq_post");
r_seq.addr = 32'h0000_0004;
r_seq.lock = 1'b0;
r_seq.start(p_sequencer.mst_seqr());
if (r_seq.rdata !== 32'hDEAD_BEEF)
`uvm_error("M3_SMOKE",
$sformatf("READBACK MISMATCH: exp=0xDEADBEEF act=0x%08h",
r_seq.rdata))
else
`uvm_info("M3_SMOKE",
$sformatf("READBACK OK: 0x%08h ✓", r_seq.rdata), UVM_LOW)
if (r_seq.resp !== R_OKAY)
`uvm_error("M3_SMOKE",
$sformatf("Readback resp not OKAY: %s",
ahb_seq_item::resp_name(r_seq.resp)))
`uvm_info("M3_SMOKE", "===== M3 Smoke Test Completed =====", UVM_LOW)
endtask
endclass : ahb_m3_smoke_vseq
class ahb_burst_writeread_vseq extends uvm_sequence;
`uvm_object_utils(ahb_burst_writeread_vseq)
`uvm_declare_p_sequencer(ahb_virtual_sequencer)
logic [31:0]  base_addr   = 32'h0000_0010;
logic [2:0]   burst_type  = B_INCR4;
burst_pat_e   pattern     = PAT_DEAD;
bit           do_write    = 1'b1;
bit           do_read     = 1'b1;
function new(string name = "ahb_burst_writeread_vseq"); super.new(name); endfunction
task body();
ahb_write_burst_seq w;
ahb_read_burst_seq  r;
int unsigned N;
logic [31:0] expected[$];
case (burst_type)
B_INCR4, B_WRAP4  : N = 4;
B_INCR8, B_WRAP8  : N = 8;
B_INCR16,B_WRAP16 : N = 16;
default           : N = 1;
endcase
if (do_write) begin
w = ahb_write_burst_seq::type_id::create("w");
w.base_addr  = base_addr;
w.burst_type = burst_type;
w.pattern    = pattern;
w.start(p_sequencer.mst_seqr());
expected = {};
for (int i = 0; i < N; i++) begin
case (pattern)
PAT_DEAD  : expected.push_back(32'hDEAD_BE00 | i);
PAT_INDEX : expected.push_back(32'hAAAA_0000 | i);
default   : expected.push_back(32'h0);
endcase
end
end
if (do_read) begin
r = ahb_read_burst_seq::type_id::create("r");
r.base_addr  = base_addr;
r.burst_type = burst_type;
r.start(p_sequencer.mst_seqr());
if (do_write && pattern != PAT_RAND) begin
for (int i = 0; i < r.rdata_out.size() && i < expected.size(); i++) begin
if (r.rdata_out[i] !== expected[i])
`uvm_error("VSEQ_CHK",
$sformatf("read-back mismatch beat=%0d exp=0x%08h got=0x%08h",
i, expected[i], r.rdata_out[i]))
end
end
end
`uvm_info("VSEQ", $sformatf("burst WR done base=0x%08h burst=%0d",
base_addr, burst_type), UVM_LOW)
endtask
endclass : ahb_burst_writeread_vseq

class ahb_retry_recover_vseq extends uvm_sequence;

`uvm_declare_p_sequencer(ahb_virtual_sequencer)
int unsigned slave_id  = 1;
bit          is_write  = 1'b0;
int unsigned num_retry = 2;
logic [31:0] base_addr = 32'h0000_0040;
logic [31:0] wr_data   = 32'hDEAD_BEEF;
function new(string name = "ahb_retry_recover_vseq"); super.new(name); endfunction
task body();
ahb_force_seq        fseq;
ahb_write_single_seq wseq;
ahb_read_single_seq  rseq;
fseq = ahb_force_seq::type_id::create("fseq");
fseq.slave_id   = slave_id;
fseq.op_split   = 1'b0;
fseq.op_retry   = 1'b1;
fseq.beat_index = 0;
fseq.num_forces = num_retry;
fork
fseq.start(p_sequencer.slv_rsp_seqr_h);
begin
if (is_write) begin
wseq = ahb_write_single_seq::type_id::create("wseq");
wseq.addr         = base_addr;
wseq.wdata        = wr_data;
wseq.max_attempts = num_retry + 1;
wseq.start(p_sequencer.mst_seqr());
end else begin
rseq = ahb_read_single_seq::type_id::create("rseq");
rseq.addr         = base_addr;
rseq.max_attempts = num_retry + 1;
rseq.start(p_sequencer.mst_seqr());
end
end

if (is_write) begin
ahb_read_single_seq verify_rd;
verify_rd = ahb_read_single_seq::type_id::create("verify_rd");
verify_rd.addr = base_addr;
verify_rd.start(p_sequencer.mst_seqr());

`uvm_error("RETRY_VSEQ",
$sformatf("DATA INTEGRITY FAIL sau retry-write: addr=0x%08h exp=0x%08h act=0x%08h",
base_addr, wr_data, verify_rd.rdata))
else
`uvm_info("RETRY_VSEQ",
$sformatf("DATA INTEGRITY OK: doc lai 0x%08h == da ghi 0x%08h",
verify_rd.rdata, wr_data), UVM_LOW)
end
`uvm_info("RETRY_VSEQ",
$sformatf("done slv=%0d dir=%s base=0x%08h num_retry=%0d",
slave_id, is_write ? "W" : "R", base_addr, num_retry), UVM_LOW)

tag, i, exp[i], r.rdata_out[i]))

endtask
task body();
`uvm_info("L2_BURST", "T2.8: INCR4->S1 then INCR8->S2 via interconnect", UVM_LOW)
run_burst(32'h0000_0040, B_INCR4, PAT_INDEX, 4, "S1_INCR4");

class ahb_lock_write_burst_seq extends ahb_base_seq;
rand logic [31:0]   base_addr;
rand logic [2:0]    burst_type;
burst_pat_e         pattern      = PAT_INDEX;
logic [31:0]        user_data[$];
bit                 abort_aware  = 1'b0;
int unsigned        max_attempts = 1;
logic [1:0]   resp;
constraint c_align { base_addr[1:0] == 2'b00; }
constraint c_dflt  { soft burst_type == B_INCR4; }
`uvm_object_utils(ahb_lock_write_burst_seq)
function new(string name = "ahb_lock_write_burst_seq"); super.new(name); endfunction
function int unsigned beats_of(logic [2:0] b);
case (b)
B_INCR4, B_WRAP4  : return 4;
B_INCR8, B_WRAP8  : return 8;
B_INCR16,B_WRAP16 : return 16;
default           : return 1;
endcase
endfunction
task body();
ahb_seq_item it;
int unsigned N;
it = ahb_seq_item::type_id::create("lock_wr_burst");
N  = beats_of(burst_type);
start_item(it);
it.dir          = AHB_WRITE;
it.addr         = base_addr;
it.trans        = T_NONSEQ;
it.burst        = burst_type;
it.size         = SZ_WORD;
it.lock         = 1'b1;
it.busreq       = 1'b1;
it.num_beats    = N;
it.abort_aware  = abort_aware;
it.max_attempts = max_attempts;
it.beats_wr     = {};
if (user_data.size() == N) begin
it.beats_wr = user_data;
end else begin
for (int i = 0; i < N; i++) begin
case (pattern)
PAT_DEAD  : it.beats_wr.push_back(32'hDEAD_BE00 | i);
PAT_INDEX : it.beats_wr.push_back(32'hAAAA_0000 | i);
default   : it.beats_wr.push_back($urandom());
endcase
end
end
it.wdata    = it.beats_wr[0];
it.exp_resp = R_OKAY;
finish_item(it);
resp = it.resp;
`uvm_info("SEQ_LOCKWR",
$sformatf("locked burst done base=0x%08h burst=%0d N=%0d resp=%s",
base_addr, burst_type, N, ahb_seq_item::resp_name(resp)),
UVM_LOW)
endtask
endclass : ahb_lock_write_burst_seq
class ahb_l3_locked_burst_vseq extends uvm_sequence;

case (pat)
PAT_DEAD : return 32'hDEAD_BE00 | i;
default  : return 32'hAAAA_0000 | i;
endcase
endfunction
task burst_wr_rd_locked(logic [31:0] baddr, burst_pat_e pat, string tag);
ahb_lock_write_burst_seq lb;
ahb_read_burst_seq       rb;
lb = ahb_lock_write_burst_seq::type_id::create({"lb_", tag});

`uvm_error("L3_LOCKB",
$sformatf("%s locked burst beat0 resp=%s != OKAY (lock loi?)",
tag, ahb_seq_item::resp_name(lb.resp)))
rb = ahb_read_burst_seq::type_id::create({"rb_", tag});
rb.base_addr  = baddr;
rb.burst_type = B_INCR4;
rb.start(p_sequencer.m1_seqr_h);
for (int i = 0; i < 4; i++)
if (rb.rdata_out[i] !== exp_beat(pat, i))
`uvm_error("L3_LOCKB",
$sformatf("%s beat %0d readback FAIL exp=0x%08h got=0x%08h",
tag, i, exp_beat(pat, i), rb.rdata_out[i]))
endtask

`uvm_info("L3_LOCKB",
"T3.5: M1 locked INCR4 + M2 xin bus -> M2 bi chan toi khi HLOCK nha",
UVM_LOW)

`uvm_error("L3_LOCKB",
$sformatf("PhaA M1 locked burst resp=%s != OKAY",
ahb_seq_item::resp_name(lbA.resp)))
end

ahb_read_single_seq  r2;
w2 = ahb_write_single_seq::type_id::create("w2");
w2.addr = m2_addr; w2.wdata = m2_data;
w2.start(p_sequencer.m2_seqr_h);
if (w2.resp !== R_OKAY)
`uvm_error("L3_LOCKB",
$sformatf("PhaA M2 write (sau lock) resp=%s != OKAY",
ahb_seq_item::resp_name(w2.resp)))
r2 = ahb_read_single_seq::type_id::create("r2");
r2.addr = m2_addr;
r2.start(p_sequencer.m2_seqr_h);
if (r2.rdata !== m2_data)
`uvm_error("L3_LOCKB",
$sformatf("PhaA M2 readback FAIL exp=0x%08h got=0x%08h",
m2_data, r2.rdata))
end
join
begin
ahb_read_burst_seq rbA;
rbA = ahb_read_burst_seq::type_id::create("rbA");
rbA.base_addr  = m1_burst_addr;
rbA.burst_type = B_INCR4;
rbA.start(p_sequencer.m1_seqr_h);
for (int i = 0; i < 4; i++)
if (rbA.rdata_out[i] !== exp_beat(PAT_INDEX, i))
`uvm_error("L3_LOCKB",
$sformatf("PhaA M1 burst beat %0d readback FAIL exp=0x%08h got=0x%08h",
i, exp_beat(PAT_INDEX, i), rbA.rdata_out[i]))
end
`uvm_info("L3_LOCKB", "PhaA OK: burst M1 nguyen ven, M2 chi lay bus sau lock",
UVM_LOW)
if (do_split_phase) begin
fork
begin : force_thr
ahb_force_seq fseq;
fseq = ahb_force_seq::type_id::create("fseqB");
fseq.slave_id   = 1;
fseq.op_split   = 1'b1;
fseq.op_retry   = 1'b0;
fseq.beat_index = 2;
fseq.num_forces = 1;
fseq.start(p_sequencer.slv_rsp_seqr_h);
end
begin : m1_lockB_thr
burst_wr_rd_locked(m1_burst_addr2, PAT_DEAD, "B");
end
join
`uvm_info("L3_LOCKB",
"PhaB OK: SPLIT bi suppress duoi HLOCK, burst hoan tat OKAY (HLOCK>SPLIT)",
UVM_LOW)
end
endtask

if (cseq.beats_resp[2] !== R_OKAY)
`uvm_error("L3_CHAIN",
$sformatf("S3 khong OKAY: %s", ahb_seq_item::resp_name(cseq.beats_resp[2])))
if (cseq.beats_resp[3] !== R_ERROR)
`uvm_error("L3_CHAIN",
$sformatf("Default khong ERROR (dung nhu spec) - got %s",
ahb_seq_item::resp_name(cseq.beats_resp[3])))
end
r1 = ahb_read_single_seq::type_id::create("chain_chk_s1");
r1.addr = s1_addr; r1.start(p_sequencer.m1_seqr_h);
if (r1.rdata !== wr_data[0])
`uvm_error("L3_CHAIN",
$sformatf("S1 readback sai exp=0x%08h got=0x%08h", wr_data[0], r1.rdata))
r2 = ahb_read_single_seq::type_id::create("chain_chk_s2");
r2.addr = s2_addr; r2.start(p_sequencer.m1_seqr_h);
if (r2.rdata !== wr_data[1])
`uvm_error("L3_CHAIN",
$sformatf("S2 readback sai exp=0x%08h got=0x%08h", wr_data[1], r2.rdata))
r3 = ahb_read_single_seq::type_id::create("chain_chk_s3");
r3.addr = s3_addr; r3.start(p_sequencer.m1_seqr_h);
if (r3.rdata !== wr_data[2])
`uvm_error("L3_CHAIN",
$sformatf("S3 readback sai exp=0x%08h got=0x%08h", wr_data[2], r3.rdata))
`uvm_info("L3_CHAIN",
"OK: pipeline S1->S2->S3->Default hoan tat, S1/S2/S3 dung du lieu, Default ERROR dung spec",
UVM_LOW)
endtask
endclass : ahb_l3_chain_pipelined_vseq

class ahb_mid_reset_traffic_vseq extends uvm_sequence;
    `uvm_object_utils(ahb_mid_reset_traffic_vseq)
    `uvm_declare_p_sequencer(ahb_virtual_sequencer)
 
    logic [31:0] addr;
    logic [31:0] wdata;
    string       tag = "midrst";
 
    function new(string name = "ahb_mid_reset_traffic_vseq");
        super.new(name);
    endfunction
 
    task body();
        ahb_write_single_seq w;
        ahb_read_single_seq  r;
 
        w = ahb_write_single_seq::type_id::create({"w_", tag});
        w.addr  = addr;
        w.wdata = wdata;
        w.start(p_sequencer.mst_seqr());
        if (w.resp !== R_OKAY)
            `uvm_error("MIDRST_SEQ",
                $sformatf("%s write @0x%08h resp=%s != OKAY",
                          tag, addr, ahb_seq_item::resp_name(w.resp)))
 
        r = ahb_read_single_seq::type_id::create({"r_", tag});
        r.addr = addr;
        r.start(p_sequencer.mst_seqr());
        if (r.rdata !== wdata)
            `uvm_error("MIDRST_SEQ",
                $sformatf("%s readback FAIL @0x%08h exp=0x%08h got=0x%08h",
                          tag, addr, wdata, r.rdata))
        else
            `uvm_info("MIDRST_SEQ",
                $sformatf("%s WR/RD @0x%08h = 0x%08h OK", tag, addr, wdata),
                UVM_LOW)
    endtask
endclass : ahb_mid_reset_traffic_vseq
