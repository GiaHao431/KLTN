class ahb_single_test extends ahb_base_test;
`uvm_component_utils(ahb_single_test)
function new(string name = "ahb_single_test", uvm_component parent = null);
super.new(name, parent);
endfunction
task run_phase(uvm_phase phase);
ahb_write_single_seq w;
ahb_read_single_seq  r;
string dir_s  = "WR";
int    base_v = 32'h0000_0004;
int    exp_v  = 0;
phase.raise_objection(this, "single_test");
phase.phase_done.set_drain_time(this, 500ns);
void'($value$plusargs("DIR=%s",   dir_s));
void'($value$plusargs("BASE=%h",  base_v));
void'($value$plusargs("EXP_INIT=%h", exp_v));
begin int unsigned m_sel = 1; void'($value$plusargs("MASTER=%d", m_sel));
`uvm_info("CFG", $sformatf("single_test: DIR=%s BASE=0x%08h EXP_INIT=0x%08h MASTER=%0d",
dir_s, base_v, exp_v, m_sel), UVM_LOW) end
#200ns;
case (dir_s)
"W" : begin
w = ahb_write_single_seq::type_id::create("w");
w.addr  = base_v;
w.wdata = 32'hDEAD_BEEF;
w.start(env.v_seqr.mst_seqr());
end
"R" : begin
r = ahb_read_single_seq::type_id::create("r");
r.addr = base_v;
r.start(env.v_seqr.mst_seqr());
if (exp_v != 0 && r.rdata !== exp_v[31:0])
`uvm_error("T1_RD",
$sformatf("initial mismatch exp=0x%08h got=0x%08h", exp_v, r.rdata))
else
`uvm_info("T1_RD", $sformatf("rdata=0x%08h OK", r.rdata), UVM_LOW)
end
"WR" : begin
w = ahb_write_single_seq::type_id::create("w");
w.addr  = base_v;
w.wdata = 32'hCAFE_BABE;
w.start(env.v_seqr.mst_seqr());
#50ns;
r = ahb_read_single_seq::type_id::create("r");
r.addr = base_v;
r.start(env.v_seqr.mst_seqr());
if (r.rdata !== 32'hCAFE_BABE)
`uvm_error("T1_WR", $sformatf("WR mismatch exp=0xCAFEBABE got=0x%08h", r.rdata))
else
`uvm_info("T1_WR", "write-read-back OK", UVM_LOW)
end
default : `uvm_error("CFG", $sformatf("unknown DIR='%s'", dir_s))
endcase
#200ns;
phase.drop_objection(this, "single_test done");
endtask
endclass : ahb_single_test
class ahb_burst_test extends ahb_base_test;
`uvm_component_utils(ahb_burst_test)
function new(string name = "ahb_burst_test", uvm_component parent = null);
super.new(name, parent);
endfunction
function logic [2:0] parse_burst(string s);
case (s)
"SINGLE" : return B_SINGLE;
"INCR"   : return B_INCR;
"WRAP4"  : return B_WRAP4;
"INCR4"  : return B_INCR4;
"WRAP8"  : return B_WRAP8;
"INCR8"  : return B_INCR8;
"WRAP16" : return B_WRAP16;
"INCR16" : return B_INCR16;
default  : begin
`uvm_warning("CFG", $sformatf("unknown BURST '%s', default INCR4", s))
return B_INCR4;
end
endcase
endfunction
function burst_pat_e parse_pattern(string s);
case (s)
"DEAD"  : return PAT_DEAD;
"INDEX" : return PAT_INDEX;
"RAND"  : return PAT_RAND;
default : return PAT_DEAD;
endcase
endfunction
task run_phase(uvm_phase phase);
ahb_burst_writeread_vseq vseq;
string  burst_s   = "INCR4";
string  dir_s     = "WR";
string  pat_s     = "DEAD";
int     base_v    = 32'h0000_0010;
phase.raise_objection(this, "burst_test");
phase.phase_done.set_drain_time(this, 500ns);
void'($value$plusargs("BURST=%s",   burst_s));
void'($value$plusargs("DIR=%s",     dir_s));
void'($value$plusargs("BASE=%h",    base_v));
void'($value$plusargs("PATTERN=%s", pat_s));
`uvm_info("CFG", $sformatf("burst_test: BURST=%s DIR=%s BASE=0x%08h PAT=%s",
burst_s, dir_s, base_v, pat_s), UVM_LOW)
#200ns;
vseq = ahb_burst_writeread_vseq::type_id::create("vseq");
vseq.base_addr  = base_v;
vseq.burst_type = parse_burst(burst_s);
vseq.pattern    = parse_pattern(pat_s);
case (dir_s)
"W"  : begin vseq.do_write = 1; vseq.do_read = 0; end
"R"  : begin vseq.do_write = 0; vseq.do_read = 1; end
"WR" : begin vseq.do_write = 1; vseq.do_read = 1; end
default : begin
`uvm_warning("CFG", $sformatf("unknown DIR '%s', default WR", dir_s))
vseq.do_write = 1; vseq.do_read = 1;
end
endcase
vseq.start(env.v_seqr);
#200ns;
phase.drop_objection(this, "burst_test done");
endtask
endclass : ahb_burst_test

class ahb_l3_retry_vs_split_test extends ahb_base_test;
`uvm_component_utils(ahb_l3_retry_vs_split_test)
function new(string name = "ahb_l3_retry_vs_split_test", uvm_component parent = null);
super.new(name, parent);
endfunction
task run_phase(uvm_phase phase);
ahb_l3_rvs_retry_vseq va;
ahb_l3_rvs_split_vseq vb;
int unsigned hs_sum_before, hs_sum_mid;
int unsigned hs_bit_mid, hs_bit_after;
int unsigned m_sel = 1;
phase.raise_objection(this, "l3_retry_vs_split");
phase.phase_done.set_drain_time(this, 500ns);
void'($value$plusargs("MASTER=%d", m_sel));
if (m_sel != 2) m_sel = 1;
`uvm_info("CFG",
$sformatf("l3_retry_vs_split: PhaA RETRY (0 mask) vs PhaB SPLIT (mask) MASTER=%0d",
m_sel), UVM_LOW)
#200ns;
hs_sum_before = env.bus_mon_agent.mon.n_hsplit_bit0_pulse
+ env.bus_mon_agent.mon.n_hsplit_bit1_pulse;
va = ahb_l3_rvs_retry_vseq::type_id::create("va");
va.start(env.v_seqr);
#100ns;
hs_sum_mid = env.bus_mon_agent.mon.n_hsplit_bit0_pulse
+ env.bus_mon_agent.mon.n_hsplit_bit1_pulse;
hs_bit_mid = (m_sel == 2) ? env.bus_mon_agent.mon.n_hsplit_bit1_pulse
: env.bus_mon_agent.mon.n_hsplit_bit0_pulse;
vb = ahb_l3_rvs_split_vseq::type_id::create("vb");
vb.start(env.v_seqr);
#200ns;
hs_bit_after = (m_sel == 2) ? env.bus_mon_agent.mon.n_hsplit_bit1_pulse
: env.bus_mon_agent.mon.n_hsplit_bit0_pulse;
if (hs_sum_mid - hs_sum_before != 0)
`uvm_error("T3_18",
$sformatf("PhaA (RETRY) co xung HSPLIT[0/1]=%0d — RETRY dang bi mask sai (cond 1/3)",
hs_sum_mid - hs_sum_before))
if (hs_bit_after - hs_bit_mid == 0)
`uvm_error("T3_18",
$sformatf("PhaB (SPLIT) KHONG co xung HSPLIT[%0d] (M%0d) — master khong duoc mask/unmask (cond 2)",
(m_sel == 2) ? 1 : 0, m_sel))
`uvm_info("T3_18",
$sformatf("PhaA HSPLIT[0/1] delta=%0d (ky vong 0) | PhaB HSPLIT[%0d](M%0d) delta=%0d (ky vong >=1)",
hs_sum_mid - hs_sum_before, (m_sel == 2) ? 1 : 0, m_sel,
hs_bit_after - hs_bit_mid), UVM_LOW)
phase.drop_objection(this, "l3_retry_vs_split done");
endtask
endclass : ahb_l3_retry_vs_split_test

class ahb_mid_reset_recovery_test extends ahb_base_test;
    `uvm_component_utils(ahb_mid_reset_recovery_test)
 
    virtual ahb_reset_ctrl_if rst_vif;
 
    function new(string name = "ahb_mid_reset_recovery_test",
                 uvm_component parent = null);
        super.new(name, parent);
    endfunction
 
    function void build_phase(uvm_phase phase);
        super.build_phase(phase);
        if (!uvm_config_db#(virtual ahb_reset_ctrl_if)::get(this, "", "rst_vif", rst_vif))
            `uvm_fatal("CFGDB", $sformatf("rst_vif MISSING for %s", get_full_name()))
        else
            `uvm_info("CFGDB", $sformatf("rst_vif OK for %s", get_full_name()), UVM_LOW)
    endfunction
 
    task run_phase(uvm_phase phase);
        ahb_mid_reset_traffic_vseq pre_vseq, post_vseq;
 
        phase.raise_objection(this, "mid_reset_recovery");
        phase.phase_done.set_drain_time(this, 500ns);
 
        // ---------------- PHA 1: van hanh binh thuong TRUOC reset ----------
        #200ns;
        pre_vseq = ahb_mid_reset_traffic_vseq::type_id::create("pre_vseq");
        pre_vseq.addr  = 32'h0000_0004;
        pre_vseq.wdata = 32'hAAAA_5555;
        pre_vseq.tag   = "pre_reset";
        pre_vseq.start(env.v_seqr);
 
        // Cho them margin de chac chan driver da ve idle that su
        // (HTRANS=IDLE, HBUSREQ=0) truoc khi kich reset.
        #200ns;
 
        // ---------------- PHA 2: kich xung HRESETn giua chung ---------------
        `uvm_info("MIDRST", "Yeu cau xung HRESETn giua chung simulation", UVM_LOW)
        @(posedge rst_vif.HCLK);
        rst_vif.reset_req = 1'b1;
        @(posedge rst_vif.HCLK);
        rst_vif.reset_req = 1'b0;
        @(posedge rst_vif.reset_ack);
        `uvm_info("MIDRST", "Reset giua chung hoan tat -- tiep tuc Pha 3", UVM_LOW)
        #100ns;
 
        // ---------------- PHA 3: van hanh binh thuong SAU reset -------------
        post_vseq = ahb_mid_reset_traffic_vseq::type_id::create("post_vseq");
        post_vseq.addr  = 32'h0000_0008;
        post_vseq.wdata = 32'h1234_5678;
        post_vseq.tag   = "post_reset";
        post_vseq.start(env.v_seqr);
        #200ns;
 
        phase.drop_objection(this, "mid_reset_recovery done");
    endtask
endclass : ahb_mid_reset_recovery_test
