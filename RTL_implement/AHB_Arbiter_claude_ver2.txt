`timescale 1ns / 1ps
//=============================================================================
// Module      : AHB_Arbiter
// Project     : AHB 2.0 Multi-master Interconnect (2 Masters, 3 Slaves)
// Language    : SystemVerilog (IEEE 1800-2012)
// Revision    : 2.0  — HMASTER encoding changed to user convention (0=M1, 1=M2)
//
// Description :
//   Bus arbiter for an AMBA AHB 2.0 multi-master system. Implements a
//   3-state Moore FSM (ST_IDLE / ST_GNT_M1 / ST_GNT_M2) with:
//     • Round-Robin fairness via `last_master_q` register
//     • HREADY-gated handover synchronisation
//     • Locked-transfer (HMASTLOCK) enforcement
//     • SPLIT response handling with per-master mask registers
//
// HMASTER Encoding Convention (v2):
//   ┌─────────────────┬──────────────────────────────────────────────┐
//   │  HMASTER value  │  Meaning                                     │
//   ├─────────────────┼──────────────────────────────────────────────┤
//   │     4'd0        │  Master 1 owns bus  (also emitted at IDLE)   │
//   │     4'd1        │  Master 2 owns bus                           │
//   │  others (≥2)    │  Reserved / illegal — MUX defaults to M1     │
//   └─────────────────┴──────────────────────────────────────────────┘
//   Rationale: 0-based index (0=M1, 1=M2) is cleaner for small-N systems.
//   IDLE state outputs 4'd0 (same as M1-granted); the MUX treats them
//   identically — when idle, M1 holds HTRANS=IDLE so the shared bus
//   appears idle regardless.
//
// Race-Condition Mitigation:
//   - All combinational logic derived exclusively from registered signals.
//   - Non-blocking (<=) in always_ff; blocking (=) in always_comb.
//   - `unique case` throughout; explicit default in every case statement.
//
//=============================================================================

module AHB_Arbiter (
    // ── System Signals ───────────────────────────────────────────────────────
    input  logic        HCLK,
    input  logic        HRESETn,          // Active-LOW asynchronous reset

    // ── Master Bus-Request & Lock Signals ────────────────────────────────────
    input  logic        HBUSREQ_M1,
    input  logic        HBUSREQ_M2,
    input  logic        HLOCK_M1,
    input  logic        HLOCK_M2,

    // ── Bus Status ───────────────────────────────────────────────────────────
    input  logic [1:0]  HTRANS_S,
    input  logic [2:0]  HBURST_S,
    input  logic        HREADY,           // Global HREADY (1 = transfer complete)
    input  logic [1:0]  HRESP_S,          // Slave response

    // ── SPLIT Completion Bus ─────────────────────────────────────────────────
    input  logic [15:0] HSPLIT_COMBINED,

    // ── Arbiter Outputs ──────────────────────────────────────────────────────
    output logic        HGRANT_M1,
    output logic        HGRANT_M2,
    output logic [3:0]  HMASTER,          // 0=M1/idle, 1=M2  (v2 encoding)
    output logic        HMASTLOCK
);

    //=========================================================================
    // ❶  Local Parameters
    //=========================================================================
    localparam logic [1:0] RESP_OKAY  = 2'b00;
    localparam logic [1:0] RESP_ERROR = 2'b01;
    localparam logic [1:0] RESP_RETRY = 2'b10;
    localparam logic [1:0] RESP_SPLIT = 2'b11;

    localparam int unsigned SPLIT_BIT_M1 = 1;
    localparam int unsigned SPLIT_BIT_M2 = 2;

    //=========================================================================
    // ❷  FSM State Type
    //=========================================================================
    typedef enum logic [1:0] {
        ST_IDLE   = 2'b00,
        ST_GNT_M1 = 2'b01,
        ST_GNT_M2 = 2'b10
    } arb_state_e;

    //=========================================================================
    // ❸  Registered Signals
    //=========================================================================
    arb_state_e  state_q;
    logic [1:0]  last_master_q;
    logic        split_mask_m1_q;
    logic        split_mask_m2_q;

    //=========================================================================
    // ❹  Combinational Signals
    //=========================================================================
    arb_state_e  state_d;
    logic        eff_req_m1;
    logic        eff_req_m2;
    logic        split_confirmed;

    //=========================================================================
    // ❺  Effective-Request & SPLIT Detection
    //=========================================================================
    assign eff_req_m1     = HBUSREQ_M1 & ~split_mask_m1_q;
    assign eff_req_m2     = HBUSREQ_M2 & ~split_mask_m2_q;
    assign split_confirmed = (HRESP_S == RESP_SPLIT) & HREADY;

    //=========================================================================
    // ❻  Next-State Logic
    //=========================================================================
    always_comb begin : NEXT_STATE_PROC
        state_d = state_q;

        unique case (state_q)

            ST_IDLE: begin
                if (eff_req_m1 && (!eff_req_m2 || last_master_q == 2'd2))
                    state_d = ST_GNT_M1;
                else if (eff_req_m2 && (!eff_req_m1 || last_master_q == 2'd1))
                    state_d = ST_GNT_M2;
            end

            ST_GNT_M1: begin
                if (split_confirmed) begin
                    state_d = eff_req_m2 ? ST_GNT_M2 : ST_IDLE;
                end else if (HLOCK_M1) begin
                    state_d = ST_GNT_M1;
                end else if (HREADY) begin
                    if (!eff_req_m1 && !eff_req_m2)
                        state_d = ST_IDLE;
                    else if (eff_req_m2 && (!eff_req_m1 || last_master_q == 2'd1))
                        state_d = ST_GNT_M2;
                    else
                        state_d = ST_GNT_M1;
                end
            end

            ST_GNT_M2: begin
                if (split_confirmed) begin
                    state_d = eff_req_m1 ? ST_GNT_M1 : ST_IDLE;
                end else if (HLOCK_M2) begin
                    state_d = ST_GNT_M2;
                end else if (HREADY) begin
                    if (!eff_req_m1 && !eff_req_m2)
                        state_d = ST_IDLE;
                    else if (eff_req_m1 && (!eff_req_m2 || last_master_q == 2'd2))
                        state_d = ST_GNT_M1;
                    else
                        state_d = ST_GNT_M2;
                end
            end

            default: state_d = ST_IDLE;

        endcase
    end : NEXT_STATE_PROC

    //=========================================================================
    // ❼  Sequential Block
    //=========================================================================
    always_ff @(posedge HCLK or negedge HRESETn) begin : SEQ_PROC

        if (!HRESETn) begin
            state_q         <= ST_IDLE;
            last_master_q   <= 2'd2;
            split_mask_m1_q <= 1'b0;
            split_mask_m2_q <= 1'b0;

        end else begin

            state_q <= state_d;

            if      (state_q == ST_GNT_M1 && state_d != ST_GNT_M1)
                last_master_q <= 2'd1;
            else if (state_q == ST_GNT_M2 && state_d != ST_GNT_M2)
                last_master_q <= 2'd2;

            if (HSPLIT_COMBINED[SPLIT_BIT_M1])
                split_mask_m1_q <= 1'b0;
            else if (state_q == ST_GNT_M1 && HRESP_S == RESP_SPLIT)
                split_mask_m1_q <= 1'b1;

            if (HSPLIT_COMBINED[SPLIT_BIT_M2])
                split_mask_m2_q <= 1'b0;
            else if (state_q == ST_GNT_M2 && HRESP_S == RESP_SPLIT)
                split_mask_m2_q <= 1'b1;

        end
    end : SEQ_PROC

    //=========================================================================
    // ❽  Output Logic  (Moore: outputs from state_q only — glitch-free)
    //
    //  ★ v2 HMASTER Encoding:
    //       ST_IDLE   → HMASTER = 4'd0  (same as M1; MUX routes M1/idle safely)
    //       ST_GNT_M1 → HMASTER = 4'd0  (0 = Master 1 selected)
    //       ST_GNT_M2 → HMASTER = 4'd1  (1 = Master 2 selected)
    //=========================================================================
    always_comb begin : OUTPUT_PROC
        // Mandatory defaults
        HGRANT_M1 = 1'b0;
        HGRANT_M2 = 1'b0;
        HMASTER   = 4'd0;   // Default = M1/idle
        HMASTLOCK = 1'b0;

        unique case (state_q)

            ST_GNT_M1: begin
                HGRANT_M1 = 1'b1;
                HGRANT_M2 = 1'b0;
                HMASTER   = 4'd0;   // ★ v2: 0 = M1 owns bus
                HMASTLOCK = HLOCK_M1;
            end

            ST_GNT_M2: begin
                HGRANT_M1 = 1'b0;
                HGRANT_M2 = 1'b1;
                HMASTER   = 4'd1;   // ★ v2: 1 = M2 owns bus
                HMASTLOCK = HLOCK_M2;
            end

            default: begin              // ST_IDLE or illegal
                HGRANT_M1 = 1'b0;
                HGRANT_M2 = 1'b0;
                HMASTER   = 4'd0;   // Idle → 0, falls to M1 safe default in MUX
                HMASTLOCK = 1'b0;
            end

        endcase
    end : OUTPUT_PROC

endmodule : AHB_Arbiter
//=============================================================================
// End of AHB_Arbiter v2
//=============================================================================
