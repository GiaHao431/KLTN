`timescale 1ns / 1ps

module ahb_arbiter (
    input  logic        HCLK,
    input  logic        HRESETn,
    
    // Tín hiệu yêu cầu từ Masters
    input  logic        HBUSREQ_M1,
    input  logic        HLOCK_M1,
    input  logic        HBUSREQ_M2,
    input  logic        HLOCK_M2,
    
    // Tín hiệu hồi tiếp từ hệ thống Slaves (đã qua Mux/OR)
    input  logic        HREADY,
    input  logic [1:0]  HRESP_S,
    input  logic [15:0] HSPLIT_COMBINED,
    
    // Tín hiệu cấp quyền và trạng thái phát ra hệ thống
    output logic        HGRANT_M1,
    output logic        HGRANT_M2,
    output logic [3:0]  HMASTER,
    output logic        HMASTLOCK
);

    // 1. Định nghĩa FSM States
    typedef enum logic [1:0] {
        IDLE   = 2'b00,
        GNT_M1 = 2'b01,
        GNT_M2 = 2'b10
    } state_t;

    state_t current_state, next_state;
    
    // Biến nhớ phục vụ phân xử Round-Robin
    logic [1:0] last_master;
    
    // 2. Thanh ghi mặt nạ (Mask) để xử lý SPLIT
    // Khi một Master bị SPLIT, mask tương ứng = 1 để loại Master đó khỏi vòng phân xử tạm thời
    logic mask_m1, mask_m2;
    
    // Tín hiệu yêu cầu thực tế (Effective Requests)
    logic eff_req_m1, eff_req_m2;
    
    assign eff_req_m1 = HBUSREQ_M1 & ~mask_m1;
    assign eff_req_m2 = HBUSREQ_M2 & ~mask_m2;

    // 3. Logic cập nhật FSM và biến nhớ last_master
    always_ff @(posedge HCLK or negedge HRESETn) begin
        if (!HRESETn) begin
            current_state <= IDLE;
            last_master   <= 2'd2; // Trạng thái Reset: Ưu tiên nhường quyền cho M1 trước 
        end else begin
            current_state <= next_state;
            
            // Cập nhật last_master khi có sự bàn giao quyền [cite: 51]
            if (current_state != next_state) begin
                if (next_state == GNT_M1)
                    last_master <= 2'd1;
                else if (next_state == GNT_M2)
                    last_master <= 2'd2;
            end
        end
    end

    // 4. Logic xử lý Mask cho SPLIT Transfer [cite: 43]
    always_ff @(posedge HCLK or negedge HRESETn) begin
        if (!HRESETn) begin
            mask_m1 <= 1'b0;
            mask_m2 <= 1'b0;
        end else begin
            // Trạng thái Mask của Master 1
            if (current_state == GNT_M1 && HRESP_S == 2'b11) 
                mask_m1 <= 1'b1; // Bị Slave trả SPLIT -> Mask [cite: 44, 45]
            else if (HSPLIT_COMBINED[1] == 1'b1) 
                mask_m1 <= 1'b0; // Slave báo hiệu đã sẵn sàng -> Unmask [cite: 46, 47]

            // Trạng thái Mask của Master 2
            if (current_state == GNT_M2 && HRESP_S == 2'b11) 
                mask_m2 <= 1'b1; // Bị Slave trả SPLIT -> Mask [cite: 44, 45]
            else if (HSPLIT_COMBINED[2] == 1'b1) 
                mask_m2 <= 1'b0; // Slave báo hiệu đã sẵn sàng -> Unmask [cite: 46, 47]
        end
    end

    // 5. Logic tổ hợp giải mã trạng thái tiếp theo (Next State Logic)
    always_comb begin
        // Mặc định giữ trạng thái hiện tại để tránh Latch
        next_state = current_state;

        case (current_state)
            IDLE: begin
                // Nếu có yêu cầu và đảm bảo tính công bằng Round-Robin [cite: 52, 56]
                if (eff_req_m1 && (!eff_req_m2 || last_master == 2'd2))
                    next_state = GNT_M1;
                else if (eff_req_m2 && (!eff_req_m1 || last_master == 2'd1))
                    next_state = GNT_M2;
            end

            GNT_M1: begin
                // A. Tước quyền NGAY LẬP TỨC nếu dính SPLIT (Không cần chờ HREADY) [cite: 55, 64]
                if (HRESP_S == 2'b11 && !HLOCK_M1) begin
                    if (eff_req_m2) next_state = GNT_M2;
                    else            next_state = IDLE;
                end 
                // B. Bàn giao bình thường: Chỉ xảy ra khi Slave đã hoàn tất giao dịch cũ (HREADY = 1) [cite: 40, 53]
                else if (HREADY && !HLOCK_M1) begin
                    if (eff_req_m2)                  next_state = GNT_M2; // M2 có yêu cầu -> Nhường bus [cite: 56]
                    else if (!eff_req_m1)            next_state = IDLE;   // Cả hai không yêu cầu -> Về IDLE [cite: 56]
                end
                // Nếu HLOCK_M1 = 1, Arbiter phớt lờ mọi yêu cầu khác và duy trì GNT_M1 [cite: 42, 54]
            end

            GNT_M2: begin
                // A. Tước quyền NGAY LẬP TỨC nếu dính SPLIT (Không cần chờ HREADY) [cite: 55, 64]
                if (HRESP_S == 2'b11 && !HLOCK_M2) begin
                    if (eff_req_m1) next_state = GNT_M1;
                    else            next_state = IDLE;
                end 
                // B. Bàn giao bình thường: Chỉ xảy ra khi Slave đã hoàn tất giao dịch cũ (HREADY = 1) [cite: 40, 53]
                else if (HREADY && !HLOCK_M2) begin
                    if (eff_req_m1)                  next_state = GNT_M1; // M1 có yêu cầu -> Nhường bus [cite: 56]
                    else if (!eff_req_m2)            next_state = IDLE;   // Cả hai không yêu cầu -> Về IDLE [cite: 56]
                end
                // Nếu HLOCK_M2 = 1, Arbiter phớt lờ mọi yêu cầu khác và duy trì GNT_M2 [cite: 42, 54]
            end
        endcase
    end

    // 6. Logic giải mã tín hiệu Output (Output Logic)
    always_comb begin
        // Khởi tạo mặc định
        HGRANT_M1 = 1'b0;
        HGRANT_M2 = 1'b0;
        HMASTER   = 4'd0;
        HMASTLOCK = 1'b0;

        case (current_state)
            GNT_M1: begin
                HGRANT_M1 = 1'b1;
                HMASTER   = 4'd1;        // Broadcast ID của Master 1 ra bus
                HMASTLOCK = HLOCK_M1;    // Kéo cờ Locked cho Slave biết nếu M1 yêu cầu khóa bus [cite: 54]
            end
            
            GNT_M2: begin
                HGRANT_M2 = 1'b1;
                HMASTER   = 4'd2;        // Broadcast ID của Master 2 ra bus
                HMASTLOCK = HLOCK_M2;    // Kéo cờ Locked cho Slave biết nếu M2 yêu cầu khóa bus [cite: 54]
            end
            
            IDLE: begin
                HMASTER   = 4'd0;        // 0 là ID dự phòng/IDLE [cite: 50]
            end
        endcase
    end

endmodule
