// =============================================================================
// MODULE 1: ahb_decoder
// =============================================================================
module ahb_decoder (
    input  logic [31:0] HADDR_S,

    output logic        HSEL_S1,
    output logic        HSEL_S2,
    output logic        HSEL_S3,
    output logic        HSEL_DEFAULT,
    output logic [1:0]  MUX_SEL
);

    always_comb begin
        HSEL_S1      = 1'b0;
        HSEL_S2      = 1'b0;
        HSEL_S3      = 1'b0;
        HSEL_DEFAULT = 1'b0;
        MUX_SEL      = 2'b00;

        case (HADDR_S[31:30])
            2'b00: begin
                HSEL_S1 = 1'b1;
                MUX_SEL = 2'b00;
            end
            2'b01: begin
                HSEL_S2 = 1'b1;
                MUX_SEL = 2'b01;
            end
            2'b10: begin
                HSEL_S3 = 1'b1;
                MUX_SEL = 2'b10;
            end
            2'b11: begin
                HSEL_DEFAULT = 1'b1;
                MUX_SEL      = 2'b11;
            end
        endcase
    end

endmodule


// =============================================================================
// MODULE 2: ahb_addr_ctrl_mux  ★ ENCODING v2 ★
//
//  Quy ước HMASTER_SEL (USER-DEFINED, 0-based):
//      4'd0 = Master 1 owns bus  (cả idle lẫn GNT_M1 đều = 0)
//      4'd1 = Master 2 owns bus
//      others → default → M1 (safe fallback)
//
//  Arbiter được sửa tương ứng: ST_GNT_M1→4'd0, ST_GNT_M2→4'd1, ST_IDLE→4'd0
// =============================================================================
module ahb_addr_ctrl_mux (
    input  logic [3:0]  HMASTER_SEL,

    input  logic [31:0] HADDR_M1,
    input  logic [1:0]  HTRANS_M1,
    input  logic        HWRITE_M1,
    input  logic [2:0]  HSIZE_M1,
    input  logic [2:0]  HBURST_M1,

    input  logic [31:0] HADDR_M2,
    input  logic [1:0]  HTRANS_M2,
    input  logic        HWRITE_M2,
    input  logic [2:0]  HSIZE_M2,
    input  logic [2:0]  HBURST_M2,

    output logic [31:0] HADDR_S,
    output logic [1:0]  HTRANS_S,
    output logic        HWRITE_S,
    output logic [2:0]  HSIZE_S,
    output logic [2:0]  HBURST_S
);

    always_comb begin
        // Safe default: idle — drive M1 bus-idle values
        HADDR_S  = HADDR_M1;
        HTRANS_S = HTRANS_M1;
        HWRITE_S = HWRITE_M1;
        HSIZE_S  = HSIZE_M1;
        HBURST_S = HBURST_M1;

        if (HMASTER_SEL == 4'd0) begin          // 4'd0 → Master 1 (cả idle + GNT_M1)
            HADDR_S  = HADDR_M1;
            HTRANS_S = HTRANS_M1;
            HWRITE_S = HWRITE_M1;
            HSIZE_S  = HSIZE_M1;
            HBURST_S = HBURST_M1;
        end else if (HMASTER_SEL == 4'd1) begin // 4'd1 → Master 2
            HADDR_S  = HADDR_M2;
            HTRANS_S = HTRANS_M2;
            HWRITE_S = HWRITE_M2;
            HSIZE_S  = HSIZE_M2;
            HBURST_S = HBURST_M2;
        end
        // else: illegal/unknown → retains safe default (M1) — no latch
    end

endmodule : ahb_addr_ctrl_mux


// =============================================================================
// MODULE 3: ahb_write_data_mux  ★ ENCODING v2: 0→M1, 1→M2 ★
// =============================================================================
module ahb_write_data_mux (
    input  logic        HCLK,
    input  logic        HRESETn,
    input  logic        HREADY,

    input  logic [3:0]  HMASTER_SEL,

    input  logic [31:0] HWDATA_M1,
    input  logic [31:0] HWDATA_M2,

    output logic [31:0] HWDATA_S
);

    logic [3:0] master_sel_data_phase;

    always_ff @(posedge HCLK or negedge HRESETn) begin
        if (!HRESETn)
            master_sel_data_phase <= 4'd0;
        else if (HREADY)
            master_sel_data_phase <= HMASTER_SEL;
    end

    always_comb begin
        HWDATA_S = HWDATA_M1;   // safe default

        if (master_sel_data_phase == 4'd0)       // 4'd0 → M1 (idle + GNT_M1)
            HWDATA_S = HWDATA_M1;
        else if (master_sel_data_phase == 4'd1)  // 4'd1 → M2
            HWDATA_S = HWDATA_M2;
    end

endmodule : ahb_write_data_mux

module ahb_read_data_mux (
    input  logic        HCLK,
    input  logic        HRESETn,
    input  logic        HREADY,

    input  logic [1:0]  MUX_SEL,

    input  logic [31:0] HRDATA_S1,
    input  logic [31:0] HRDATA_S2,
    input  logic [31:0] HRDATA_S3,

    output logic [31:0] HRDATA_M
);

    logic [1:0] mux_sel_data_phase;

    always_ff @(posedge HCLK or negedge HRESETn) begin
        if (!HRESETn)
            mux_sel_data_phase <= 2'b00;
        else if (HREADY)
            mux_sel_data_phase <= MUX_SEL;
    end

    always_comb begin
        case (mux_sel_data_phase)
            2'b00:   HRDATA_M = HRDATA_S1;
            2'b01:   HRDATA_M = HRDATA_S2;
            2'b10:   HRDATA_M = HRDATA_S3;
            2'b11:   HRDATA_M = 32'h0000_0000;
            default: HRDATA_M = 32'h0000_0000;
        endcase
    end

endmodule

// =============================================================================
// MODULE 7: AHB_Arbiter
// =============================================================================
module AHB_Arbiter (
    input  logic        HCLK,
    input  logic        HRESETn,

    input  logic        HBUSREQ_M1,
    input  logic        HBUSREQ_M2,
    input  logic        HLOCK_M1,
    input  logic        HLOCK_M2,

    input  logic [1:0]  HTRANS_S,
    input  logic [2:0]  HBURST_S,
    input  logic        HREADY,
    input  logic [1:0]  HRESP_S,

    input  logic [15:0] HSPLIT_COMBINED,

    output logic        HGRANT_M1,
    output logic        HGRANT_M2,
    output logic [3:0]  HMASTER,
    output logic        HMASTLOCK
);

    localparam logic [1:0] RESP_OKAY  = 2'b00;
    localparam logic [1:0] RESP_ERROR = 2'b01;
    localparam logic [1:0] RESP_RETRY = 2'b10;
    localparam logic [1:0] RESP_SPLIT = 2'b11;

    localparam int unsigned SPLIT_BIT_M1 = 1;
    localparam int unsigned SPLIT_BIT_M2 = 2;

    typedef enum logic [1:0] {
        ST_IDLE   = 2'b00,
        ST_GNT_M1 = 2'b01,
        ST_GNT_M2 = 2'b10
    } arb_state_e;

    arb_state_e  state_q;
    logic [1:0]  last_master_q;
    logic        split_mask_m1_q;
    logic        split_mask_m2_q;

    arb_state_e  state_d;
    logic        eff_req_m1;
    logic        eff_req_m2;
    logic        split_confirmed;

    assign eff_req_m1 = HBUSREQ_M1 & ~split_mask_m1_q;
    assign eff_req_m2 = HBUSREQ_M2 & ~split_mask_m2_q;
    assign split_confirmed = (HRESP_S == RESP_SPLIT) & HREADY;

    always_comb begin : NEXT_STATE_PROC
        state_d = state_q;

        unique case (state_q)
            ST_IDLE: begin
                if (eff_req_m1 && (!eff_req_m2 || last_master_q == 2'd2))
                    state_d = ST_GNT_M1;
                else if (eff_req_m2 && (!eff_req_m1 || last_master_q == 2'd1))
                    state_d = ST_GNT_M2;
            end

            ST_GNT_M1: begin
                if (split_confirmed) begin
                    state_d = eff_req_m2 ? ST_GNT_M2 : ST_IDLE;
                end else if (HLOCK_M1) begin
                    state_d = ST_GNT_M1;
                end else if (HREADY) begin
                    if (!eff_req_m1 && !eff_req_m2)
                        state_d = ST_IDLE;
                    else if (eff_req_m2 && (!eff_req_m1 || last_master_q == 2'd1))
                        state_d = ST_GNT_M2;
                    else
                        state_d = ST_GNT_M1;
                end
            end

            ST_GNT_M2: begin
                if (split_confirmed) begin
                    state_d = eff_req_m1 ? ST_GNT_M1 : ST_IDLE;
                end else if (HLOCK_M2) begin
                    state_d = ST_GNT_M2;
                end else if (HREADY) begin
                    if (!eff_req_m1 && !eff_req_m2)
                        state_d = ST_IDLE;
                    else if (eff_req_m1 && (!eff_req_m2 || last_master_q == 2'd2))
                        state_d = ST_GNT_M1;
                    else
                        state_d = ST_GNT_M2;
                end
            end

            default: state_d = ST_IDLE;
        endcase
    end : NEXT_STATE_PROC

    always_ff @(posedge HCLK or negedge HRESETn) begin : SEQ_PROC
        if (!HRESETn) begin
            state_q         <= ST_IDLE;
            last_master_q   <= 2'd2;
            split_mask_m1_q <= 1'b0;
            split_mask_m2_q <= 1'b0;
        end else begin
            state_q <= state_d;

            if      (state_q == ST_GNT_M1 && state_d != ST_GNT_M1)
                last_master_q <= 2'd1;
            else if (state_q == ST_GNT_M2 && state_d != ST_GNT_M2)
                last_master_q <= 2'd2;

            if (HSPLIT_COMBINED[SPLIT_BIT_M1])
                split_mask_m1_q <= 1'b0;
            else if (state_q == ST_GNT_M1 && HRESP_S == RESP_SPLIT)
                split_mask_m1_q <= 1'b1;

            if (HSPLIT_COMBINED[SPLIT_BIT_M2])
                split_mask_m2_q <= 1'b0;
            else if (state_q == ST_GNT_M2 && HRESP_S == RESP_SPLIT)
                split_mask_m2_q <= 1'b1;
        end
    end : SEQ_PROC

    always_comb begin : OUTPUT_PROC
        HGRANT_M1 = 1'b0;
        HGRANT_M2 = 1'b0;
        HMASTER   = 4'd0;
        HMASTLOCK = 1'b0;

        unique case (state_q)
            ST_GNT_M1: begin
                HGRANT_M1 = 1'b1;
                HGRANT_M2 = 1'b0;
                HMASTER   = 4'd0;   // 0 = M1 owns bus (user encoding)
                HMASTLOCK = HLOCK_M1;
            end
            ST_GNT_M2: begin
                HGRANT_M1 = 1'b0;
                HGRANT_M2 = 1'b1;
                HMASTER   = 4'd1;   // 1 = M2 owns bus (user encoding)
                HMASTLOCK = HLOCK_M2;
            end
            default: begin
                HGRANT_M1 = 1'b0;
                HGRANT_M2 = 1'b0;
                HMASTER   = 4'd0;
                HMASTLOCK = 1'b0;
            end
        endcase
    end : OUTPUT_PROC

endmodule : AHB_Arbiter


// =============================================================================
// MODULE 8: AHB_Slave
// =============================================================================
module AHB_Slave (
    input  logic        HCLK,
    input  logic        HRESETn,
    input  logic        HREADY_IN,
    input  logic        HSELx,
    input  logic [31:0] HADDR,
    input  logic [1:0]  HTRANS,
    input  logic        HWRITE,
    input  logic [2:0]  HSIZE,
    input  logic [2:0]  HBURST,
    input  logic [31:0] HWDATA,
    output logic [31:0] HRDATA,
    input  logic [3:0]  HMASTER,
    input  logic        HMASTLOCK,
    output logic [15:0] HSPLITx,
    output logic        HREADYOUT,
    output logic [1:0]  HRESP
);

localparam logic [1:0]
    T_IDLE   = 2'b00, T_BUSY   = 2'b01,
    T_NONSEQ = 2'b10, T_SEQ    = 2'b11;

localparam logic [1:0]
    R_OKAY = 2'b00, R_ERROR = 2'b01,
    R_RETRY = 2'b10, R_SPLIT = 2'b11;

localparam logic [2:0]
    B_SINGLE = 3'b000, B_INCR   = 3'b001,
    B_WRAP4  = 3'b010, B_INCR4  = 3'b011,
    B_WRAP8  = 3'b100, B_INCR8  = 3'b101,
    B_WRAP16 = 3'b110, B_INCR16 = 3'b111;

localparam logic [2:0]
    ST_IDLE      = 3'b000, ST_ACTIVE    = 3'b001,
    ST_AGAIN     = 3'b010, ST_LITTLE    = 3'b011,
    ST_TIME_PASS = 3'b111,
    ST_WBURST    = 3'b100, ST_RBURST    = 3'b101;

logic [2:0] ps_slave, ns_slave;
logic [7:0] local_addr, local_addr_base;
logic       local_write;
logic [2:0] local_burst, local_size;
logic [5:0] beat_cnt;
logic [3:0] split_master_reg;
logic [31:0] memory_slave [0:255];

initial begin : mem_init
    for (int i = 0; i < 256; i++) memory_slave[i] = 32'd0;
    memory_slave[0] = 32'd10;
    memory_slave[1] = 32'd20;
    memory_slave[2] = 32'd30;
end

function automatic logic [7:0] next_burst_addr (
    input logic [7:0] base, input logic [7:0] cur, input logic [2:0] burst
);
    logic [7:0] lin;
    lin = cur + 8'h01;
    case (burst)
        B_WRAP4  : return {base[7:2], lin[1:0]};
        B_WRAP8  : return {base[7:3], lin[2:0]};
        B_WRAP16 : return {base[7:4], lin[3:0]};
        default  : return lin;
    endcase
endfunction

// SEQ BLOCK 1: State Register
always_ff @(posedge HCLK or negedge HRESETn) begin
    if (!HRESETn) ps_slave <= ST_IDLE;
    else          ps_slave <= ns_slave;
end

// SEQ BLOCK 2: Address-Phase Pipeline Latch
always_ff @(posedge HCLK or negedge HRESETn) begin
    if (!HRESETn) begin
        local_addr      <= 8'h00;
        local_addr_base <= 8'h00;
        local_write     <= 1'b0;
        local_burst     <= B_SINGLE;
        local_size      <= 3'b010;
    end else begin
        if (HSELx && HREADY_IN && (HTRANS == T_NONSEQ) &&
            (ps_slave == ST_IDLE || ps_slave == ST_AGAIN)) begin
            local_addr      <= HADDR[9:2];
            local_addr_base <= HADDR[9:2];
            local_write     <= HWRITE;
            local_burst     <= HBURST;
            local_size      <= HSIZE;
        end
        else if ((ps_slave == ST_WBURST || ps_slave == ST_RBURST) &&
                  beat_cnt > 6'd1) begin
            local_addr <= next_burst_addr(local_addr_base, local_addr, local_burst);
        end
    end
end

// SEQ BLOCK 3: Beat Counter
always_ff @(posedge HCLK or negedge HRESETn) begin
    if (!HRESETn) begin
        beat_cnt <= 6'd0;
    end else begin
        case (ps_slave)
            ST_ACTIVE : begin
                case (local_burst)
                    B_SINGLE           : beat_cnt <= 6'd1;
                    B_INCR             : beat_cnt <= 6'd32;
                    B_WRAP4, B_INCR4   : beat_cnt <= 6'd4;
                    B_WRAP8, B_INCR8   : beat_cnt <= 6'd8;
                    B_WRAP16, B_INCR16 : beat_cnt <= 6'd16;
                    default            : beat_cnt <= 6'd1;
                endcase
            end
            ST_WBURST, ST_RBURST : begin
                if (beat_cnt > 6'd0) beat_cnt <= beat_cnt - 6'd1;
            end
            default : ;
        endcase
    end
end

// SEQ BLOCK 4: SRAM Write
always_ff @(posedge HCLK) begin
    if (ps_slave == ST_WBURST)
        memory_slave[local_addr] <= HWDATA;
end

// SEQ BLOCK 5: SPLIT Master-ID Latch
always_ff @(posedge HCLK or negedge HRESETn) begin
    if (!HRESETn)
        split_master_reg <= 4'h0;
    else if (ps_slave == ST_RBURST && local_burst == B_SINGLE && local_addr == 8'h00)
        split_master_reg <= HMASTER;
end

// COMB BLOCK: Next-State & Output Logic
always_comb begin : fsm_comb
    ns_slave  = ps_slave;
    HREADYOUT = 1'b1;
    HRESP     = R_OKAY;
    HRDATA    = 32'h0000_0000;
    HSPLITx   = 16'h0000;

    case (ps_slave)

        ST_IDLE : begin
            HREADYOUT = 1'b1;
            HRESP     = R_OKAY;
            if (HSELx && HREADY_IN && (HTRANS == T_NONSEQ))
                ns_slave = ST_ACTIVE;
        end

        ST_ACTIVE : begin
            HREADYOUT = 1'b0;
            HRESP     = R_OKAY;
            ns_slave  = local_write ? ST_WBURST : ST_RBURST;
        end

        ST_AGAIN : begin
            HREADYOUT = 1'b0;
            HRESP     = R_RETRY;
            ns_slave  = ST_ACTIVE;
        end

        ST_WBURST : begin
            HRESP = R_OKAY;
            case (local_burst)
                B_SINGLE : begin
                    HREADYOUT = 1'b1;
                    ns_slave  = ST_IDLE;
                end
                B_INCR, B_WRAP4, B_INCR4, B_WRAP8, B_INCR8, B_WRAP16, B_INCR16 : begin
                    HREADYOUT = 1'b1;
                    ns_slave  = (beat_cnt <= 6'd1) ? ST_IDLE : ST_WBURST;
                end
                default : begin
                    HREADYOUT = 1'b1;
                    ns_slave  = ST_IDLE;
                end
            endcase
        end

        ST_RBURST : begin
            HRDATA = memory_slave[local_addr];
            case (local_burst)
                B_SINGLE : begin
                    if (local_addr == 8'h00) begin
                        HREADYOUT = 1'b0;
                        HRESP     = R_SPLIT;
                        HRDATA    = 32'h0000_0000;
                        ns_slave  = ST_LITTLE;
                    end else begin
                        HREADYOUT = 1'b1;
                        HRESP     = R_OKAY;
                        ns_slave  = ST_IDLE;
                    end
                end
                B_INCR, B_WRAP4, B_INCR4, B_WRAP8, B_INCR8, B_WRAP16, B_INCR16 : begin
                    HREADYOUT = 1'b1;
                    HRESP     = R_OKAY;
                    ns_slave  = (beat_cnt <= 6'd1) ? ST_IDLE : ST_RBURST;
                end
                default : begin
                    HREADYOUT = 1'b1;
                    HRESP     = R_OKAY;
                    ns_slave  = ST_IDLE;
                end
            endcase
        end

        ST_LITTLE : begin
            HREADYOUT = 1'b1;
            HRESP     = R_SPLIT;
            if (!HMASTLOCK) begin
                HSPLITx  = 16'h0001 << split_master_reg;
                ns_slave = ST_IDLE;
            end else begin
                HSPLITx  = 16'h0000;
                ns_slave = ST_TIME_PASS;
            end
        end

        ST_TIME_PASS : begin
            HREADYOUT = 1'b1;
            HRESP     = R_OKAY;
            if (!HMASTLOCK) begin
                HSPLITx  = 16'h0001 << split_master_reg;
                ns_slave = ST_IDLE;
            end else begin
                HSPLITx  = 16'h0000;
                ns_slave = ST_TIME_PASS;
            end
        end

        default : begin
            ns_slave  = ST_IDLE;
            HREADYOUT = 1'b1;
            HRESP     = R_OKAY;
        end

    endcase
end : fsm_comb

endmodule


// =============================================================================
// MODULE 9: AHB_Default_Slave
// =============================================================================
module AHB_Default_Slave (
    input  logic        HCLK,
    input  logic        HRESETn,
    input  logic        HSEL_DEFAULT,
    input  logic [1:0]  HTRANS,
    output logic        HREADYOUT_DEFAULT,
    output logic [1:0]  HRESP_DEFAULT
);

localparam logic [1:0]
    T_IDLE = 2'b00, T_BUSY = 2'b01,
    T_NONSEQ = 2'b10, T_SEQ = 2'b11;

localparam logic [1:0]
    R_OKAY = 2'b00, R_ERROR = 2'b01;

typedef enum logic [1:0] {
    ST_IDLE       = 2'b00,
    ST_ERROR_CYC1 = 2'b01,
    ST_ERROR_CYC2 = 2'b10
} ahb_def_state_t;

ahb_def_state_t ps, ns;

logic active_transfer;
assign active_transfer = HSEL_DEFAULT &&
                         (HTRANS == T_NONSEQ || HTRANS == T_SEQ);

always_ff @(posedge HCLK or negedge HRESETn) begin
    if (!HRESETn) ps <= ST_IDLE;
    else          ps <= ns;
end

always_comb begin : fsm_comb
    ns                 = ps;
    HREADYOUT_DEFAULT  = 1'b1;
    HRESP_DEFAULT      = R_OKAY;

    case (ps)
        ST_IDLE : begin
            HREADYOUT_DEFAULT = 1'b1;
            HRESP_DEFAULT     = R_OKAY;
            if (active_transfer)
                ns = ST_ERROR_CYC1;
            else
                ns = ST_IDLE;
        end
        ST_ERROR_CYC1 : begin
            HREADYOUT_DEFAULT = 1'b0;
            HRESP_DEFAULT     = R_ERROR;
            ns                = ST_ERROR_CYC2;
        end
        ST_ERROR_CYC2 : begin
            HREADYOUT_DEFAULT = 1'b1;
            HRESP_DEFAULT     = R_ERROR;
            ns                = ST_IDLE;
        end
        default : begin
            HREADYOUT_DEFAULT = 1'b1;
            HRESP_DEFAULT     = R_OKAY;
            ns                = ST_IDLE;
        end
    endcase
end : fsm_comb

endmodule

// =============================================================================
// MODULE 11: ahb_master_bfm_l2  (Layer 2 BFM)
// =============================================================================

`define T_IDLE   2'b00
`define T_BUSY   2'b01
`define T_NONSEQ 2'b10
`define T_SEQ    2'b11

`define B_SINGLE 3'b000
`define B_INCR   3'b001
`define B_WRAP4  3'b010
`define B_INCR4  3'b011

`define R_OKAY   2'b00
`define R_ERROR  2'b01
`define R_RETRY  2'b10
`define R_SPLIT  2'b11

`define SZ_WORD  3'b010

module ahb_master_bfm_l2 (
    input  logic        HCLK,
    output logic        HRESETn,

    output logic        HBUSREQ,
    output logic        HLOCK,
    input  logic        HGRANT,

    output logic [31:0] HADDR,
    output logic [1:0]  HTRANS,
    output logic        HWRITE,
    output logic [2:0]  HSIZE,
    output logic [2:0]  HBURST,

    output logic [31:0] HWDATA,
    input  logic [31:0] HRDATA,

    input  logic        HREADY,
    input  logic [1:0]  HRESP
);

    initial begin
        HRESETn  = 1'b1;
        HBUSREQ  = 1'b0;
        HLOCK    = 1'b0;
        HADDR    = 32'h0;
        HTRANS   = `T_IDLE;
        HWRITE   = 1'b0;
        HSIZE    = `SZ_WORD;
        HBURST   = `B_SINGLE;
        HWDATA   = 32'h0;
    end

    task automatic ahb_reset(input int cycles = 5);
        HRESETn = 1'b0;
        HBUSREQ = 1'b0;
        HLOCK   = 1'b0;
        HADDR   = 32'h0;
        HTRANS  = `T_IDLE;
        HWRITE  = 1'b0;
        HWDATA  = 32'h0;
        repeat (cycles) @(posedge HCLK);
        #1;
        HRESETn = 1'b1;
        @(posedge HCLK);
    endtask

    task automatic ahb_idle();
        @(posedge HCLK); #1;
        HTRANS = `T_IDLE;
        HADDR  = 32'h0;
    endtask

    task automatic ensure_grant();
        HBUSREQ = 1'b1;
        if (!HGRANT) begin
            @(posedge HCLK);
            while (!HGRANT) @(posedge HCLK);
        end
    endtask

    task automatic ahb_write_single(
        input  logic [31:0] addr,
        input  logic [31:0] data,
        ref    logic        ok
    );
        ensure_grant();

        @(posedge HCLK); #1;
        HADDR  = addr;
        HTRANS = `T_NONSEQ;
        HWRITE = 1'b1;
        HSIZE  = `SZ_WORD;
        HBURST = `B_SINGLE;

        @(posedge HCLK);
        while (!HREADY) @(posedge HCLK);
        #1;

        HWDATA = data;
        HTRANS = `T_IDLE;

        @(posedge HCLK);
        while (!HREADY) @(posedge HCLK);

        ok = (HRESP == `R_OKAY) ? 1'b1 : 1'b0;

        #1;
        HWRITE = 1'b0;
        HWDATA = 32'h0;
    endtask

    task automatic ahb_read_single(
        input  logic [31:0] addr,
        ref    logic [31:0] data,
        ref    logic        ok
    );
        ensure_grant();

        @(posedge HCLK); #1;
        HADDR  = addr;
        HTRANS = `T_NONSEQ;
        HWRITE = 1'b0;
        HSIZE  = `SZ_WORD;
        HBURST = `B_SINGLE;

        @(posedge HCLK);
        while (!HREADY) @(posedge HCLK);
        #1;
        HTRANS = `T_IDLE;

        @(posedge HCLK);
        while (!HREADY) @(posedge HCLK);

        data = HRDATA;
        ok   = (HRESP == `R_OKAY) ? 1'b1 : 1'b0;
        #1;
    endtask

    task automatic ahb_write_burst_incr4(
        input  logic [31:0] base_addr,
        input  logic [31:0] wdata [0:3],
        ref    logic        ok
    );
        logic [31:0] cur_addr;
        ok = 1'b1;
        ensure_grant();

        @(posedge HCLK); #1;
        HADDR    = base_addr;
        HTRANS   = `T_NONSEQ;
        HWRITE   = 1'b1;
        HSIZE    = `SZ_WORD;
        HBURST   = `B_INCR4;
        cur_addr = base_addr;

        for (int i = 0; i < 3; i++) begin
            @(posedge HCLK);
            while (!HREADY) @(posedge HCLK);
            #1;
            HWDATA   = wdata[i];
            cur_addr = cur_addr + 32'd4;
            HADDR    = cur_addr;
            HTRANS   = `T_SEQ;
            if (HRESP != `R_OKAY) ok = 1'b0;
        end

        @(posedge HCLK);
        while (!HREADY) @(posedge HCLK);
        #1;
        HWDATA = wdata[3];
        HTRANS = `T_IDLE;
        HADDR  = 32'h0;
        if (HRESP != `R_OKAY) ok = 1'b0;

        @(posedge HCLK);
        while (!HREADY) @(posedge HCLK);
        if (HRESP != `R_OKAY) ok = 1'b0;

        #1;
        HWRITE = 1'b0;
        HWDATA = 32'h0;
    endtask

    task automatic ahb_read_burst_incr4(
        input  logic [31:0] base_addr,
        ref    logic [31:0] rdata [0:3],
        ref    logic        ok
    );
        logic [31:0] cur_addr;
        ok = 1'b1;
        ensure_grant();

        @(posedge HCLK); #1;
        HADDR    = base_addr;
        HTRANS   = `T_NONSEQ;
        HWRITE   = 1'b0;
        HSIZE    = `SZ_WORD;
        HBURST   = `B_INCR4;
        cur_addr = base_addr;

        for (int i = 0; i < 3; i++) begin
            @(posedge HCLK);
            while (!HREADY) @(posedge HCLK);
            #1;
            rdata[i] = HRDATA;
            if (HRESP != `R_OKAY) ok = 1'b0;
            cur_addr = cur_addr + 32'd4;
            HADDR    = cur_addr;
            HTRANS   = `T_SEQ;
        end

        @(posedge HCLK);
        while (!HREADY) @(posedge HCLK);
        #1;
        rdata[3] = HRDATA;
        HTRANS   = `T_IDLE;
        HADDR    = 32'h0;
        if (HRESP != `R_OKAY) ok = 1'b0;

        @(posedge HCLK);
    endtask

endmodule : ahb_master_bfm_l2

        check("T2.1 HGRANT_M1 asserted",
              HGRANT_M1 == 1'b1, $sformatf("HGRANT_M1=%0b", HGRANT_M1));
        check("T2.1 HMASTER=0 (M1 encoding)",
              HMASTER == 4'd0, $sformatf("HMASTER=%0d", HMASTER));
        $display("");

        // =================================================================
        // T2.2 — TC_L2_M1_READ_S1
        // =================================================================
        $display("=== T2.2: TC_L2_M1_READ_S1 ===");
        $display("    M1 Read Slave 1 (0x0000_0010) — verify T2.1 data");

        u_bfm_m1.ahb_read_single(32'h0000_0010, rdata, ok);

        check("T2.2 Read OKAY",
              ok == 1'b1, "via read_data_mux");
        check("T2.2 Data Integrity",
              rdata === 32'hDEAD_BEEF,
              $sformatf("Exp=0xDEADBEEF Got=0x%08X", rdata));
        $display("");

        // =================================================================
        // T2.3 — TC_L2_DEFAULT_SLAVE_ERR
        // =================================================================
        $display("=== T2.3: TC_L2_DEFAULT_SLAVE_ERR ===");
        $display("    M1 Write to 0xC000_0000 → ERROR expected");

        fork
            begin : err_mon
                logic sel_def, err_seen;
                sel_def = 0; err_seen = 0;
                repeat (15) begin
                    @(posedge HCLK);
                    if (HSEL_DEFAULT === 1'b1) sel_def = 1;
                    if (HRESP_M === 2'b01)     err_seen = 1;
                end
                check("T2.3 HSEL_DEFAULT seen",
                      sel_def, "Decoder selected Default Slave");
                check("T2.3 ERROR on HRESP_M",
                      err_seen, "resp_mux routes HRESP_DEFAULT");
            end
            begin : err_xfer
                u_bfm_m1.ahb_write_single(32'hC000_0000, 32'hBAD0_CAFE, ok);
                check("T2.3 ERROR detected (ok=0)",
                      ok == 1'b0,
                      $sformatf("HRESP_M=%02b", HRESP_M));
            end
        join

        u_bfm_m1.ahb_idle();
        u_bfm_m1.ahb_idle();
        $display("");

        // =================================================================
        // T2.4 — TC_L2_M1_WRITE_S2
        // =================================================================
        $display("=== T2.4: TC_L2_M1_WRITE_S2 ===");
        $display("    M1 Write 0xCAFE_1234 → Slave 2 (0x4000_0010)");

        u_bfm_m1.ahb_write_single(32'h4000_0010, 32'hCAFE_1234, ok);
        check("T2.4 Write S2 OKAY", ok == 1'b1, "Decoder→S2");

        u_bfm_m1.ahb_read_single(32'h4000_0010, rdata, ok);
        check("T2.4 Read S2 Verify",
              ok == 1'b1 && rdata === 32'hCAFE_1234,
              $sformatf("Exp=0xCAFE1234 Got=0x%08X", rdata));
        $display("");

        // =================================================================
        // T2.5 — TC_L2_M1_WRITE_S3
        // =================================================================
        $display("=== T2.5: TC_L2_M1_WRITE_S3 ===");
        $display("    M1 Write 0xBEEF_5678 → Slave 3 (0x8000_0010)");

        u_bfm_m1.ahb_write_single(32'h8000_0010, 32'hBEEF_5678, ok);
        check("T2.5 Write S3 OKAY", ok == 1'b1, "Decoder→S3");

        u_bfm_m1.ahb_read_single(32'h8000_0010, rdata, ok);
        check("T2.5 Read S3 Verify",
              ok == 1'b1 && rdata === 32'hBEEF_5678,
              $sformatf("Exp=0xBEEF5678 Got=0x%08X", rdata));
        $display("");

        // =================================================================
        // T2.6 — TC_L2_HREADY_GLOBAL
        // =================================================================
        $display("=== T2.6: TC_L2_HREADY_GLOBAL ===");
        $display("    Monitor HREADY_GLOBAL during write to Slave 1");

        fork
            begin : hrdy_mon
                logic saw_low, saw_recover;
                saw_low = 0; saw_recover = 0;
                repeat (20) begin
                    @(posedge HCLK);
                    if (HREADY_GLOBAL === 1'b0) saw_low = 1;
                    if (saw_low && HREADY_GLOBAL === 1'b1) saw_recover = 1;
                end
                check("T2.6 HREADY_GLOBAL dipped to 0",
                      saw_low, "Wait-state propagated");
                check("T2.6 HREADY_GLOBAL recovered to 1",
                      saw_recover, "Pipeline resumed");
            end
            begin : hrdy_xfer
                u_bfm_m1.ahb_write_single(32'h0000_0020, 32'h1111_2222, ok);
                check("T2.6 Write OKAY", ok == 1'b1, "Feedback loop works");
            end
        join

        check("T2.6 HREADY===HREADY_GLOBAL",
              HREADY === HREADY_GLOBAL,
              $sformatf("HREADY=%0b HREADY_GLOBAL=%0b", HREADY, HREADY_GLOBAL));
        $display("");

        // =================================================================
        // T2.7 — TC_L2_CROSS_SLAVE_RAW
        // =================================================================
        $display("=== T2.7: TC_L2_CROSS_SLAVE_RAW ===");

        $display("    Step 1: Write 0xAAAA_BBBB → S1 (0x0000_0030)");
        u_bfm_m1.ahb_write_single(32'h0000_0030, 32'hAAAA_BBBB, ok);
        check("T2.7 Write S1 OKAY", ok == 1'b1, "");

        $display("    Step 2: Read S2 seed (0x4000_0008)");
        u_bfm_m1.ahb_read_single(32'h4000_0008, rdata, ok);
        check("T2.7 Read S2 OKAY", ok == 1'b1, "Cross-slave");
        check("T2.7 S2 seed=30",
              rdata === 32'd30,
              $sformatf("Exp=30 Got=%0d", rdata));

        $display("    Step 3: Re-read S1 — no contamination");
        u_bfm_m1.ahb_read_single(32'h0000_0030, rdata, ok);
        check("T2.7 S1 data preserved",
              rdata === 32'hAAAA_BBBB,
              $sformatf("Exp=0xAAAABBBB Got=0x%08X", rdata));
        $display("");

        // =================================================================
        // T2.8 — TC_L2_BURST_VIA_INTC
        // =================================================================
        $display("=== T2.8: TC_L2_BURST_VIA_INTC ===");
        $display("    INCR4 Write 4 words to S1 (base=0x0000_0040)");

        wdata_burst[0] = 32'h1111_0001;
        wdata_burst[1] = 32'h2222_0002;
        wdata_burst[2] = 32'h3333_0003;
        wdata_burst[3] = 32'h4444_0004;

        u_bfm_m1.ahb_write_burst_incr4(32'h0000_0040, wdata_burst, ok);
        check("T2.8 INCR4 Write OKAY", ok == 1'b1, "4 beats via Intc");

        $display("    INCR4 Read 4 words from S1 (base=0x0000_0040)");
        u_bfm_m1.ahb_read_burst_incr4(32'h0000_0040, rdata_burst, ok);
        check("T2.8 INCR4 Read OKAY", ok == 1'b1, "4 beats via Intc");
        check("T2.8 Beat[0]",
              rdata_burst[0] === 32'h1111_0001,
              $sformatf("Exp=0x11110001 Got=0x%08X", rdata_burst[0]));
        check("T2.8 Beat[1]",
              rdata_burst[1] === 32'h2222_0002,
              $sformatf("Exp=0x22220002 Got=0x%08X", rdata_burst[1]));
        check("T2.8 Beat[2]",
              rdata_burst[2] === 32'h3333_0003,
              $sformatf("Exp=0x33330003 Got=0x%08X", rdata_burst[2]));
        check("T2.8 Beat[3]",
              rdata_burst[3] === 32'h4444_0004,
              $sformatf("Exp=0x44440004 Got=0x%08X", rdata_burst[3]));
        $display("");

        // =================================================================
        // Summary
        // =================================================================
        #50;
        $display("================================================================");
        $display(" LAYER 2 SIMULATION COMPLETE");
        $display("----------------------------------------------------------------");
        $display(" PASS : %0d", pass_count);
        $display(" FAIL : %0d", fail_count);
        $display(" TOTAL: %0d", pass_count + fail_count);
        if (fail_count == 0)
            $display(" RESULT: *** ALL LAYER 2 TESTS PASSED ***");
        else
            $display(" RESULT: *** %0d TEST(S) FAILED ***", fail_count);
        $display("================================================================");
        $finish;
    end
