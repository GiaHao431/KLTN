`timescale 1ns / 1ps
//=============================================================================
// Module   : ahb_master_bfm
// Purpose  : AHB 2.0 Bus Functional Model — Layer 1 Unit Test Driver
//
// Vai trò  : Thay thế hoàn toàn RTL Master. Đây là một non-synthesizable
//            module chứa các task mô phỏng hành vi Master AHB chuẩn.
//            Testbench sẽ gọi các task này để phát stimulus.
//
// Tasks được cung cấp:
//   • ahb_reset()           — giữ reset trong N chu kỳ rồi release
//   • ahb_write_single()    — ghi 1 word vào địa chỉ bất kỳ
//   • ahb_read_single()     — đọc 1 word từ địa chỉ bất kỳ
//   • ahb_write_burst_incr4() — ghi 4 word liên tiếp (INCR4)
//   • ahb_read_burst_incr4()  — đọc 4 word liên tiếp (INCR4)
//   • ahb_idle()            — phát 1 chu kỳ IDLE
//
// Giao thức AHB pipeline được tuân thủ đầy đủ:
//   Address Phase (T)   → HADDR, HTRANS, HWRITE, HSIZE, HBURST lên bus
//   Data Phase    (T+1) → HWDATA hợp lệ; sample HRDATA/HRESP/HREADY
//=============================================================================

// HTRANS encoding (AHB IHI0011A Table 3-1)
`define T_IDLE   2'b00
`define T_BUSY   2'b01
`define T_NONSEQ 2'b10
`define T_SEQ    2'b11

// HBURST encoding (AHB IHI0011A Table 3-2)
`define B_SINGLE 3'b000
`define B_INCR   3'b001
`define B_WRAP4  3'b010
`define B_INCR4  3'b011
`define B_WRAP8  3'b100
`define B_INCR8  3'b101
`define B_WRAP16 3'b110
`define B_INCR16 3'b111

// HRESP encoding
`define R_OKAY   2'b00
`define R_ERROR  2'b01
`define R_RETRY  2'b10
`define R_SPLIT  2'b11

// HSIZE encoding
`define SZ_BYTE     3'b000
`define SZ_HALFWORD 3'b001
`define SZ_WORD     3'b010

module ahb_master_bfm (
    //── System ────────────────────────────────────────────────────────────────
    input  logic        HCLK,
    output logic        HRESETn,

    //── Address & Control (BFM drives → DUT receives) ────────────────────────
    output logic [31:0] HADDR,
    output logic [1:0]  HTRANS,
    output logic        HWRITE,
    output logic [2:0]  HSIZE,
    output logic [2:0]  HBURST,
    output logic [3:0]  HMASTER,     // ID của Master này (hardcoded = 4'd1)
    output logic        HMASTLOCK,

    //── Data (BFM drives HWDATA; DUT drives HRDATA) ───────────────────────────
    output logic [31:0] HWDATA,
    input  logic [31:0] HRDATA,

    //── Response (DUT drives → BFM reads) ────────────────────────────────────
    input  logic        HREADYOUT,   // Slave/Interconnect ready
    input  logic [1:0]  HRESP,

    //── Slave select (BFM drives directly for Unit Test) ─────────────────────
    output logic        HSELx
);

    //=========================================================================
    // Khởi tạo giá trị mặc định an toàn (bus idle state)
    //=========================================================================
    initial begin
        HRESETn    = 1'b1;   // reset chưa kích hoạt
        HADDR      = 32'h0;
        HTRANS     = `T_IDLE;
        HWRITE     = 1'b0;
        HSIZE      = `SZ_WORD;
        HBURST     = `B_SINGLE;
        HMASTER    = 4'd1;   // BFM này đại diện Master 1
        HMASTLOCK  = 1'b0;
        HWDATA     = 32'h0;
        HSELx      = 1'b0;
    end

    //=========================================================================
    // TASK 1: ahb_reset
    // Mô tả : Kích hoạt reset bất đồng bộ trong `cycles` chu kỳ rồi release.
    // Dùng  : Luôn gọi đầu tiên trong mọi test case.
    //=========================================================================
    task automatic ahb_reset(input int cycles = 5);
        HRESETn = 1'b0;   // kéo reset xuống thấp
        HADDR   = 32'h0;
        HTRANS  = `T_IDLE;
        HWRITE  = 1'b0;
        HWDATA  = 32'h0;
        HSELx   = 1'b0;
        repeat (cycles) @(posedge HCLK);
        #1;               // nhỏ hơn 1 chu kỳ để tránh race với posedge
        HRESETn = 1'b1;   // release reset
        @(posedge HCLK);  // chờ 1 chu kỳ ổn định
    endtask

    //=========================================================================
    // TASK 2: ahb_idle
    // Mô tả : Phát 1 chu kỳ IDLE (không có giao dịch).
    //=========================================================================
    task automatic ahb_idle();
        @(posedge HCLK); #1;
        HTRANS = `T_IDLE;
        HADDR  = 32'h0;
        HSELx  = 1'b0;
    endtask

    //=========================================================================
    // TASK 3: ahb_write_single
    // Mô tả : Thực hiện 1 giao dịch ghi đơn (SINGLE WRITE) chuẩn AHB.
    //
    // Pipeline flow:
    //   Chu kỳ T   (Address Phase): phát HADDR, HTRANS=NONSEQ, HWRITE=1
    //   Chu kỳ T+1 (Data Phase)  : phát HWDATA, chờ HREADYOUT=1
    //
    // Tham số:
    //   addr  — địa chỉ byte 32-bit cần ghi
    //   data  — giá trị 32-bit cần ghi
    //   ok    — output: 1 nếu HRESP=OKAY, 0 nếu lỗi
    //=========================================================================
    task automatic ahb_write_single(
        input  logic [31:0] addr,
        input  logic [31:0] data,
        ref logic        ok
    );
        // ── Address Phase ──────────────────────────────────────────────────
        @(posedge HCLK); #1;
        HSELx  = 1'b1;
        HADDR  = addr;
        HTRANS = `T_NONSEQ;
        HWRITE = 1'b1;
        HSIZE  = `SZ_WORD;
        HBURST = `B_SINGLE;

        // ── Data Phase ─────────────────────────────────────────────────────
        // Chờ Slave sẵn sàng (HREADYOUT=1) — xử lý wait states
        @(posedge HCLK); #1;
        HWDATA = data;
        HTRANS = `T_IDLE;  // sau SINGLE, trở về IDLE
        HSELx  = 1'b0;

        // Chờ Slave hoàn tất (có thể có nhiều wait state)
        while (!HREADYOUT) @(posedge HCLK);

        // Ghi nhận kết quả
        ok = (HRESP == `R_OKAY) ? 1'b1 : 1'b0;

        // Dọn dẹp bus sau khi xong
        @(posedge HCLK); #1;
        HWRITE = 1'b0;
        HWDATA = 32'h0;
    endtask

    //=========================================================================
    // TASK 4: ahb_read_single
    // Mô tả : Thực hiện 1 giao dịch đọc đơn (SINGLE READ) chuẩn AHB.
    //
    // Pipeline flow:
    //   Chu kỳ T   (Address Phase): phát HADDR, HTRANS=NONSEQ, HWRITE=0
    //   Chu kỳ T+1 (Data Phase)  : sample HRDATA sau khi HREADYOUT=1
    //
    // Lưu ý SPLIT: nếu địa chỉ[9:2] == 0, Slave sẽ phát SPLIT response.
    //   Task này xử lý trường hợp đó bằng cách chờ thêm.
    //
    // Tham số:
    //   addr  — địa chỉ byte 32-bit cần đọc
    //   data  — output: dữ liệu đọc được
    //   ok    — output: 1=OKAY, 0=lỗi (ERROR/SPLIT)
    //=========================================================================
    task automatic ahb_read_single(
        input  logic [31:0] addr,
        ref logic [31:0] data,
        ref logic        ok
    );
        // ── Address Phase ──────────────────────────────────────────────────
        @(posedge HCLK); #1;
        HSELx  = 1'b1;
        HADDR  = addr;
        HTRANS = `T_NONSEQ;
        HWRITE = 1'b0;
        HSIZE  = `SZ_WORD;
        HBURST = `B_SINGLE;

        // ── Data Phase ─────────────────────────────────────────────────────
        @(posedge HCLK); #1;
        HTRANS = `T_IDLE;
        HSELx  = 1'b0;

        // Chờ HREADYOUT=1 (bao gồm cả wait state và chu kỳ 1 của SPLIT)
        while (!HREADYOUT) @(posedge HCLK);

        // Capture data và response tại cạnh HREADYOUT=1
        data = HRDATA;
        ok   = (HRESP == `R_OKAY) ? 1'b1 : 1'b0;

        @(posedge HCLK); #1;
    endtask

    //=========================================================================
    // TASK 5: ahb_write_burst_incr4
    // Mô tả : Ghi 4 word liên tiếp theo kiểu INCR4 burst.
    //
    // Pipeline flow chuẩn AHB:
    //   Beat 0: Address Phase NONSEQ tại addr
    //   Beat 1: Address Phase SEQ   tại addr+4  + Data Phase beat0
    //   Beat 2: Address Phase SEQ   tại addr+8  + Data Phase beat1
    //   Beat 3: Address Phase SEQ   tại addr+12 + Data Phase beat2
    //   Beat 4: IDLE                            + Data Phase beat3
    //
    // Tham số:
    //   base_addr — địa chỉ bắt đầu (word-aligned)
    //   wdata     — mảng 4 word dữ liệu cần ghi
    //   ok        — 1 nếu tất cả beats đều OKAY
    //=========================================================================
    task automatic ahb_write_burst_incr4(
        input  logic [31:0] base_addr,
        input  logic [31:0] wdata [0:3],
        ref logic        ok
    );
        logic [31:0] cur_addr;
        ok = 1'b1;

        // ── Beat 0: Address Phase ──────────────────────────────────────────
        @(posedge HCLK); #1;
        HSELx     = 1'b1;
        HADDR     = base_addr;
        HTRANS    = `T_NONSEQ;
        HWRITE    = 1'b1;
        HSIZE     = `SZ_WORD;
        HBURST    = `B_INCR4;
        cur_addr  = base_addr;

        // ── Beat 1–3: SEQ Address Phase + Data Phase beat trước ───────────
        for (int i = 0; i < 3; i++) begin
            @(posedge HCLK); #1;
            // Chờ nếu slave chèn wait state
            while (!HREADYOUT) begin
                @(posedge HCLK); #1;
            end
            // Phát dữ liệu của beat hiện tại (i)
            HWDATA    = wdata[i];
            // Cập nhật địa chỉ sang beat tiếp theo
            cur_addr  = cur_addr + 32'd4;
            HADDR     = cur_addr;
            HTRANS    = `T_SEQ;

            if (HRESP != `R_OKAY) ok = 1'b0;
        end

        // ── Beat cuối: kết thúc burst, phát dữ liệu beat 3 ───────────────
        @(posedge HCLK); #1;
        while (!HREADYOUT) begin
            @(posedge HCLK); #1;
        end
        HWDATA = wdata[3];
        HTRANS = `T_IDLE;
        HSELx  = 1'b0;
        HADDR  = 32'h0;
        if (HRESP != `R_OKAY) ok = 1'b0;

        // Chờ data phase beat cuối hoàn tất
        @(posedge HCLK); #1;
        while (!HREADYOUT) begin
            @(posedge HCLK); #1;
        end
        if (HRESP != `R_OKAY) ok = 1'b0;

        // Dọn dẹp
        HWRITE = 1'b0;
        HWDATA = 32'h0;
    endtask

    //=========================================================================
    // TASK 6: ahb_read_burst_incr4
    // Mô tả : Đọc 4 word liên tiếp theo kiểu INCR4 burst.
    //
    // Tham số:
    //   base_addr — địa chỉ bắt đầu (word-aligned)
    //   rdata     — output: mảng 4 word dữ liệu đọc được
    //   ok        — 1 nếu tất cả beats đều OKAY
    //=========================================================================
    task automatic ahb_read_burst_incr4(
        input  logic [31:0] base_addr,
        ref logic [31:0] rdata [0:3],
        ref logic        ok
    );
        logic [31:0] cur_addr;
        ok = 1'b1;

        // ── Beat 0: Address Phase ──────────────────────────────────────────
        @(posedge HCLK); #1;
        HSELx    = 1'b1;
        HADDR    = base_addr;
        HTRANS   = `T_NONSEQ;
        HWRITE   = 1'b0;
        HSIZE    = `SZ_WORD;
        HBURST   = `B_INCR4;
        cur_addr = base_addr;

        // ── Beat 1–3: SEQ Address Phase + capture HRDATA beat trước ───────
        for (int i = 0; i < 3; i++) begin
            @(posedge HCLK); #1;
            while (!HREADYOUT) begin
                @(posedge HCLK); #1;
            end
            // Capture dữ liệu của beat i (data phase của beat i)
            rdata[i] = HRDATA;
            if (HRESP != `R_OKAY) ok = 1'b0;

            // Phát địa chỉ beat tiếp theo
            cur_addr = cur_addr + 32'd4;
            HADDR    = cur_addr;
            HTRANS   = `T_SEQ;
        end

        // ── Kết thúc burst: capture dữ liệu beat 3 ────────────────────────
        @(posedge HCLK); #1;
        while (!HREADYOUT) begin
            @(posedge HCLK); #1;
        end
        rdata[3] = HRDATA;
        HTRANS   = `T_IDLE;
        HSELx    = 1'b0;
        HADDR    = 32'h0;
        if (HRESP != `R_OKAY) ok = 1'b0;

        // Chờ data phase beat cuối
        @(posedge HCLK); #1;
    endtask

endmodule : ahb_master_bfm
